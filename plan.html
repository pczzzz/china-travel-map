⚙ AI 智能规划设置

AI 规划走 OpenAI 兼容接口（/chat/completions）。
· 本地体验：保持默认，用 python tools/serve.py 启动站点，密钥放服务端，前端零密钥。
· 线上使用：部署一个免费的 Cloudflare Worker 当网关（见下方步骤），把 Worker 地址填到「网关地址」。

没有网关？用 Cloudflare Worker 免费搭一个（约 5 分钟）

1) 打开 dash.cloudflare.com → Workers & Pages → 创建 Worker（免费额度足够个人用）。
2) 把下面的代码整个粘贴进去 → 部署 → 得到地址 https://你的名字.workers.dev/llm。
3) 在该 Worker「设置 → 变量」里配三个变量：OPENAI_KEY=你的模型服务商密钥、OPENAI_BASE=https://api.deepseek.com/v1（或通义等，注意以 /v1 结尾）、OPENAI_MODEL=deepseek-chat。
4) 把 Worker 地址填到上面的「网关地址」，模型填对应模型名，API Key 留空即可。

export default {
  async fetch(req, env) {
    if (req.method === "OPTIONS") {
      return new Response("", { status: 204, headers: cors() });
    }
    const url = new URL(req.url);
    if (url.pathname !== "/llm" || req.method !== "POST") {
      return new Response("Not Found", { status: 404 });
    }
    let body;
    try { body = await req.json(); } catch (e) { return new Response("bad json", { status: 400 }); }
    const base = env.OPENAI_BASE || "https://api.openai.com/v1";
    const model = body.model || env.OPENAI_MODEL || "gpt-4o-mini";
    const key = env.OPENAI_KEY || "";
    const r = await fetch(base + "/chat/completions", {
      method: "POST",
      headers: { "Content-Type": "application/json", "Authorization": "Bearer " + key },
      body: JSON.stringify({ model: model, messages: body.messages, temperature: body.temperature, max_tokens: body.max_tokens || 4000 })
    });
    const txt = await r.text();
    return new Response(txt, { status: r.status, headers: Object.assign({ "Content-Type": "application/json" }, cors()) });
  }
};
function cors() {
  return {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Headers": "Content-Type, Authorization",
    "Access-Control-Allow-Methods": "POST, OPTIONS"
  };
}
